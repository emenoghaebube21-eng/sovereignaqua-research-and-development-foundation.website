/* ========================================================== NAVIGATION.JS SovereignAqua Research & Development Foundation Version 2.2 Responsibilities: - Desktop/mobile navigation - Mobile drawer - Desktop dropdown - Mobile dropdown - Search panel - Announcement dismissal - Escape-key handling - Outside-click handling - Responsive state management - Active navigation state ========================================================== */ "use strict"; /* ========================================================== INITIALIZE NAVIGATION ========================================================== */ export function initNavigation() { const navigation = document.querySelector( ".sa-navigation" ); if (!navigation) { return; } initAnnouncement(); initMobileNavigation(); initDesktopDropdown(); initMobileDropdown(); initSearch(); initNavigationLinks(); initResponsiveState(); } /* ========================================================== ANNOUNCEMENT BAR ========================================================== */ function initAnnouncement() { const announcement = document.querySelector( ".sa-announcement-bar" ); const closeButton = document.querySelector( "[data-close-announcement]" ); if ( !announcement || !closeButton ) { return; } closeButton.addEventListener( "click", () => { announcement.classList.add( "is-hidden" ); announcement.setAttribute( "aria-hidden", "true" ); } ); } /* ========================================================== MOBILE NAVIGATION ========================================================== */ function initMobileNavigation() { const toggle = document.querySelector( "[data-mobile-toggle]" ); const drawer = document.querySelector( "[data-mobile-drawer]" ); const closeButton = document.querySelector( "[data-mobile-close]" ); if (!toggle || !drawer) { return; } const openDrawer = () => { drawer.classList.add( "is-open" ); drawer.setAttribute( "aria-hidden", "false" ); toggle.setAttribute( "aria-expanded", "true" ); toggle.setAttribute( "aria-label", "Close mobile menu" ); document.body.classList.add( "menu-open" ); }; const closeDrawer = () => { drawer.classList.remove( "is-open" ); drawer.setAttribute( "aria-hidden", "true" ); toggle.setAttribute( "aria-expanded", "false" ); toggle.setAttribute( "aria-label", "Open mobile menu" ); document.body.classList.remove( "menu-open" ); closeMobileDropdowns(); }; toggle.addEventListener( "click", event => { event.preventDefault(); const isOpen = toggle.getAttribute( "aria-expanded" ) === "true"; if (isOpen) { closeDrawer(); } else { openDrawer(); } } ); if (closeButton) { closeButton.addEventListener( "click", closeDrawer ); } drawer .querySelectorAll( "a[data-nav-link]" ) .forEach( link => { link.addEventListener( "click", closeDrawer ); } ); document.addEventListener( "keydown", event => { if ( event.key !== "Escape" ) { return; } const isOpen = toggle.getAttribute( "aria-expanded" ) === "true"; if (!isOpen) { return; } closeDrawer(); toggle.focus(); } ); document.addEventListener( "click", event => { const isOpen = toggle.getAttribute( "aria-expanded" ) === "true"; if (!isOpen) { return; } if ( drawer.contains( event.target ) || toggle.contains( event.target ) ) { return; } closeDrawer(); } ); } /* ========================================================== DESKTOP RESEARCH DROPDOWN ========================================================== */ function initDesktopDropdown() { const trigger = document.querySelector( "[data-dropdown-trigger]" ); const menu = document.querySelector( "[data-dropdown-menu]" ); if (!trigger || !menu) { return; } trigger.addEventListener( "click", event => { event.preventDefault(); const isOpen = trigger.getAttribute( "aria-expanded" ) === "true"; closeDesktopDropdowns(); if (!isOpen) { trigger.setAttribute( "aria-expanded", "true" ); menu.classList.add( "is-open" ); } } ); trigger.addEventListener( "keydown", event => { if ( event.key === "ArrowDown" ) { event.preventDefault(); trigger.click(); const firstItem = menu.querySelector( "a[data-nav-link]" ); if (firstItem) { firstItem.focus(); } } } ); menu .querySelectorAll( "a[data-nav-link]" ) .forEach( link => { link.addEventListener( "keydown", event => { if ( event.key === "Escape" ) { closeDesktopDropdowns(); trigger.focus(); } } ); } ); } /* ========================================================== CLOSE DESKTOP DROPDOWNS ========================================================== */ function closeDesktopDropdowns() { document .querySelectorAll( "[data-dropdown-trigger]" ) .forEach( trigger => { trigger.setAttribute( "aria-expanded", "false" ); } ); document .querySelectorAll( "[data-dropdown-menu]" ) .forEach( menu => { menu.classList.remove( "is-open" ); } ); } /* ========================================================== MOBILE RESEARCH DROPDOWN ========================================================== */ function initMobileDropdown() { const triggers = document.querySelectorAll( "[data-mobile-dropdown-trigger]" ); triggers.forEach( trigger => { const menu = trigger.parentElement ?.querySelector( "[data-mobile-dropdown-menu]" ); if (!menu) { return; } trigger.addEventListener( "click", event => { event.preventDefault(); const isOpen = trigger.getAttribute( "aria-expanded" ) === "true"; trigger.setAttribute( "aria-expanded", String(!isOpen) ); menu.style.display = isOpen ? "none" : ""; menu.classList.toggle( "is-open", !isOpen ); } ); } ); } /* ========================================================== CLOSE MOBILE DROPDOWNS ========================================================== */ function closeMobileDropdowns() { document .querySelectorAll( "[data-mobile-dropdown-trigger]" ) .forEach( trigger => { trigger.setAttribute( "aria-expanded", "false" ); } ); document .querySelectorAll( "[data-mobile-dropdown-menu]" ) .forEach( menu => { menu.style.display = "none"; menu.classList.remove( "is-open" ); } ); } /* ========================================================== SEARCH ========================================================== */ function initSearch() { const toggle = document.querySelector( "[data-search-toggle]" ); const panel = document.querySelector( "[data-search-panel]" ); const overlay = document.querySelector( "[data-search-overlay]" ); const close = document.querySelector( "[data-search-close]" ); const input = document.querySelector( "[data-search-input]" ); const form = document.querySelector( "[data-search-form]" ); if ( !toggle || !panel ) { return; } const openSearch = () => { panel.classList.add( "is-open" ); if (overlay) { overlay.classList.add( "is-visible" ); overlay.setAttribute( "aria-hidden", "false" ); } toggle.setAttribute( "aria-expanded", "true" ); if (input) { window.setTimeout( () => input.focus(), 50 ); } }; const closeSearch = () => { panel.classList.remove( "is-open" ); if (overlay) { overlay.classList.remove( "is-visible" ); overlay.setAttribute( "aria-hidden", "true" ); } toggle.setAttribute( "aria-expanded", "false" ); }; toggle.setAttribute( "aria-expanded", "false" ); toggle.addEventListener( "click", event => { event.preventDefault(); const isOpen = toggle.getAttribute( "aria-expanded" ) === "true"; if (isOpen) { closeSearch(); } else { openSearch(); } } ); if (close) { close.addEventListener( "click", closeSearch ); } if (overlay) { overlay.addEventListener( "click", closeSearch ); } document.addEventListener( "keydown", event => { if ( event.key === "Escape" ) { closeSearch(); } } ); if (form) { form.addEventListener( "submit", event => { /* Search backend is not defined by the supplied navigation markup. Prevent accidental page submission until the project's search endpoint is established. */ event.preventDefault(); const query = input ? input.value.trim() : ""; if (!query) { if (input) { input.focus(); } return; } /* Do not invent a search endpoint here. The query is intentionally left available for the future search implementation. */ console.info( "[SovereignAqua] Search query:", query ); } ); } } /* ========================================================== NAVIGATION LINKS ========================================================== */ function initNavigationLinks() { const links = document.querySelectorAll( "[data-nav-link]" ); if (!links.length) { return; } const currentPath = normalizePath( window.location.pathname ); links.forEach( link => { const href = link.getAttribute( "href" ); if ( !href || href.startsWith("#") ) { return; } const linkUrl = resolveUrl( href ); if (!linkUrl) { return; } const linkPath = normalizePath( linkUrl.pathname ); if ( linkPath === currentPath ) { link.classList.add( "is-current" ); link.setAttribute( "aria-current", "page" ); } } ); } /* ========================================================== RESPONSIVE STATE ========================================================== */ function initResponsiveState() { const mobileToggle = document.querySelector( "[data-mobile-toggle]" ); const mobileDrawer = document.querySelector( "[data-mobile-drawer]" ); if ( !mobileToggle || !mobileDrawer ) { return; } const handleResize = () => { /* Mobile navigation is replaced by the desktop navigation at the large-screen breakpoint. */ if ( window.innerWidth >= 992 ) { mobileDrawer.classList.remove( "is-open" ); mobileDrawer.setAttribute( "aria-hidden", "true" ); mobileToggle.setAttribute( "aria-expanded", "false" ); mobileToggle.setAttribute( "aria-label", "Open mobile menu" ); document.body.classList.remove( "menu-open" ); closeMobileDropdowns(); } }; window.addEventListener( "resize", handleResize, { passive: true } ); handleResize(); } /* ========================================================== URL HELPERS ========================================================== */ function normalizePath( path ) { if (!path) { return "/"; } const normalized = path .split("?")[0] .split("#")[0] .replace( /\/+$/, "" ); return normalized || "/"; } function resolveUrl( href ) { try { return new URL( href, window.location.origin ); } catch ( error ) { return null; } }
